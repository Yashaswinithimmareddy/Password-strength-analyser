Database Schema Zero-Knowledge Verification

Proof of defensive architecture: The database schema stores ONLY aggregate parameters. 

CREATE TABLE analyses (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    analyzed_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    score INTEGER NOT NULL,
    classification TEXT NOT NULL,
    password_length INTEGER NOT NULL,
    unique_character_ratio REAL NOT NULL,
    theoretical_entropy REAL NOT NULL,
    adjusted_entropy REAL NOT NULL,
    weakness_count INTEGER NOT NULL DEFAULT 0
    -- NO password column! NO hash column! NO user PII!
);

CREATE TABLE findings (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    analysis_id INTEGER NOT NULL,
    finding_type TEXT NOT NULL,
    severity TEXT NOT NULL,
    description TEXT NOT NULL
);
